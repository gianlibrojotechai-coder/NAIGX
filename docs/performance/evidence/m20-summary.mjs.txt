import fs from "node:fs";

const rows = fs
  .readFileSync(process.argv[2], "utf8")
  .trim()
  .split("\n")
  .filter(Boolean)
  .map((l) => JSON.parse(l));

// Nearest-rank percentile — the same definition src/db/metrics.ts uses.
const pct = (v, p) => {
  if (!v.length) return null;
  const s = [...v].sort((a, b) => a - b);
  return s[Math.max(0, Math.ceil((p / 100) * s.length) - 1)];
};
const f = (x) => (x === null || x === undefined ? "—" : x.toFixed(1));

const completed = rows.filter((r) => r.status === "completed");
const withArt = completed.filter((r) => r.serverFirstArtifactS !== null);
const clientArt = completed.filter((r) => r.clientFirstArtifactS !== null);
const spend = rows.reduce((a, r) => a + r.costUsd, 0);
const calls = rows.reduce((a, r) => a + r.providerCalls, 0);

console.log(
  `runs=${rows.length} completed=${completed.length} failed=${rows.length - completed.length} ` +
    `withArtifact=${withArt.length} spend=$${spend.toFixed(4)} calls=${calls}`,
);
console.log(
  `NFR-001 first artifact (server, M-16 formula): p50=${f(pct(withArt.map((r) => r.serverFirstArtifactS), 50))}s ` +
    `p95=${f(pct(withArt.map((r) => r.serverFirstArtifactS), 95))}s n=${withArt.length}  [budget 15 / 40]`,
);
console.log(
  `NFR-001 first artifact (client-observed, 400ms poll): p50=${f(pct(clientArt.map((r) => r.clientFirstArtifactS), 50))}s ` +
    `p95=${f(pct(clientArt.map((r) => r.clientFirstArtifactS), 95))}s n=${clientArt.length}`,
);
console.log(
  `NFR-002 full completion (server, M-16 formula): p50=${f(pct(completed.map((r) => r.serverFullS), 50))}s ` +
    `p95=${f(pct(completed.map((r) => r.serverFullS), 95))}s n=${completed.length}  [budget 60 / 120]`,
);
console.log(
  `NFR-002 full completion (client-observed): p50=${f(pct(completed.map((r) => r.clientCompleteS), 50))}s ` +
    `p95=${f(pct(completed.map((r) => r.clientCompleteS), 95))}s n=${completed.length}`,
);

const byType = {};
for (const r of rows) (byType[r.type ?? "?"] ??= []).push(r);
for (const [t, rs] of Object.entries(byType)) {
  const c = rs.filter((r) => r.status === "completed");
  const arts = c.filter((r) => r.serverFirstArtifactS !== null);
  console.log(
    `  ${t}: n=${rs.length} completed=${c.length} ` +
      `full p50=${f(pct(c.map((r) => r.serverFullS), 50))}s max=${c.length ? f(Math.max(...c.map((r) => r.serverFullS))) : "—"}s ` +
      `firstArt p50=${f(pct(arts.map((r) => r.serverFirstArtifactS), 50))}s max=${arts.length ? f(Math.max(...arts.map((r) => r.serverFirstArtifactS))) : "—"}s ` +
      `cost/run=$${(rs.reduce((a, r) => a + r.costUsd, 0) / rs.length).toFixed(4)}`,
  );
}
console.log(
  "completed runs with NO artifact:",
  completed
    .filter((r) => r.serverFirstArtifactS === null)
    .map((r) => `${r.caseId}#${r.run}(${r.verdict ?? "-"})`)
    .join(", ") || "none",
);
console.log(
  "failed runs:",
  rows
    .filter((r) => r.status !== "completed")
    .map((r) => `${r.caseId}#${r.run}:${r.status}`)
    .join(", ") || "none",
);
console.log(
  "degraded runs:",
  rows.filter((r) => r.degraded).map((r) => `${r.caseId}#${r.run}`).join(", ") || "none",
);
